4. Technical Implementation

Data Processing Flow

Process Description: After entering the application, the system first obtains location or allows users to select tour spots. The frontend submits spot information to the backend. The system retrieves relevant content from the cultural database and AR resource library. After data matching and processing, the system renders three types of feedback: historical information popup, AR navigation directions, and check-in completion. Finally, user tour behavior data is saved and the process ends.

GPS Location Manager with Fallback

class LocationManager {
  async getCurrentPosition(options = {}) {
    return new Promise((resolve, reject) => {
      if (!navigator.geolocation) {
        return this._handleFallback(reject);
      }
      
      const timeout = setTimeout(() => {
        this._handleFallback(reject);
      }, 10000);
      
      navigator.geolocation.getCurrentPosition(
        (position) => {
          clearTimeout(timeout);
          resolve(position);
        },
        (error) => {
          clearTimeout(timeout);
          this._handleError(error, reject);
        },
        { ...options, timeout: 10000 }
      );
    });
  }

  async _handleFallback(reject) {
    const fallbackCoords = {
      latitude: 31.3237,
      longitude: 120.6205
    };
    resolve({
      coords: fallbackCoords,
      timestamp: Date.now()
    });
  }

  _handleError(error, reject) {
    const errorTypes = {
      1: 'Permission denied',
      2: 'Position unavailable',
      3: 'Timeout'
    };
    reject(new Error(errorTypes[error.code] || 'Unknown error'));
  }
}

AR Marker Rendering

class ARMarkerRenderer {
  constructor(scene) {
    this.scene = scene;
    this.markers = [];
    this.maxDistance = 50;
  }

  addMarker(id, position, content) {
    const marker = { id, position, content, entity: null };
    this.markers.push(marker);
    this._renderMarker(marker);
  }

  _renderMarker(marker) {
    const entity = document.createElement('a-entity');
    entity.setAttribute('gps-entity-place', {
      latitude: marker.position.latitude,
      longitude: marker.position.longitude
    });
    entity.setAttribute('text', {
      value: marker.content,
      color: '#2c3e50',
      align: 'center'
    });
    entity.setAttribute('scale', '1 1 1');
    
    this.scene.appendChild(entity);
    marker.entity = entity;
  }

  updateMarkers(userPosition) {
    this.markers.forEach(marker => {
      const distance = this._calculateDistance(
        userPosition, marker.position
      );
      
      if (marker.entity) {
        marker.entity.object3D.visible = distance <= this.maxDistance;
      }
    });
  }

  _calculateDistance(pos1, pos2) {
    const R = 6371;
    const dLat = (pos2.latitude - pos1.latitude) * Math.PI / 180;
    const dLon = (pos2.longitude - pos1.longitude) * Math.PI / 180;
    const a = Math.sin(dLat/2) * Math.sin(dLat/2) +
              Math.cos(pos1.latitude * Math.PI / 180) * 
              Math.cos(pos2.latitude * Math.PI / 180) *
              Math.sin(dLon/2) * Math.sin(dLon/2);
    return R * (2 * Math.atan2(Math.sqrt(a), Math.sqrt(1-a)));
  }
}

Check-in Manager

class CheckinManager {
  constructor() {
    this.checkins = this._loadCheckins();
  }

  _loadCheckins() {
    try {
      const data = localStorage.getItem('ar_checkins');
      return data ? JSON.parse(data) : [];
    } catch {
      return [];
    }
  }

  _saveCheckins() {
    localStorage.setItem('ar_checkins', JSON.stringify(this.checkins));
  }

  async capturePhoto() {
    return new Promise((resolve, reject) => {
      const stream = await navigator.mediaDevices.getUserMedia({
        video: { facingMode: 'environment' }
      });
      
      const video = document.createElement('video');
      video.srcObject = stream;
      video.play();
      
      setTimeout(() => {
        const canvas = document.createElement('canvas');
        canvas.width = video.videoWidth;
        canvas.height = video.videoHeight;
        const ctx = canvas.getContext('2d');
        ctx.drawImage(video, 0, 0);
        
        stream.getTracks().forEach(track => track.stop());
        resolve(canvas.toDataURL('image/jpeg', 0.8));
      }, 100);
    });
  }

  addCheckin(location, photoData) {
    const checkin = {
      id: Date.now(),
      location,
      photo: photoData,
      timestamp: new Date().toISOString()
    };
    this.checkins.push(checkin);
    this._saveCheckins();
    return checkin;
  }

  getCheckins() {
    return [...this.checkins];
  }
}
